# claude-app-server Phase 1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `apps/claude-app-server/` に TypeScript ベースの Codex App Server 互換 JSON-RPC サーバのスケルトンを作り、`claude-app-server` コマンドで起動できて `initialize` リクエストにダミー応答を返せる状態にする。

**Architecture:** 純 TypeScript (ESM) + tsc。`src/jsonrpc/` に transport（stdio JSONL）と dispatcher（method ルーティング）を分離。`src/handlers/` に各 method ハンドラ。`src/bin.ts` は commander で CLI 引数をパースして `src/server.ts` の `runServer()` を呼ぶ。

**Tech Stack:** TypeScript / Node.js 20+ / ESM / tsc / vitest / commander / oxlint / oxfmt / pnpm workspace

**Spec:** [`docs/superpowers/specs/2026-05-11-cas-phase1-design.md`](../specs/2026-05-11-cas-phase1-design.md)

```typescript
export type Id = number | string;

export interface Request<P = unknown> {
  id: Id;
  method: string;
  params?: P;
  jsonrpc?: "2.0";
}

export interface Notification<P = unknown> {
  method: string;
  params?: P;
  jsonrpc?: "2.0";
}

export interface SuccessResponse<R = unknown> {
  id: Id;
  result: R;
  jsonrpc?: "2.0";
}

export interface ErrorResponse {
  id: Id | null;
  error: { code: number; message: string; data?: unknown };
  jsonrpc?: "2.0";
}

export type Response<R = unknown> = SuccessResponse<R> | ErrorResponse;

export type IncomingMessage = Request | Notification;
export type OutgoingMessage = Response | Notification;

export const ErrorCode = {
  ParseError: -32700,
  InvalidRequest: -32600,
  MethodNotFound: -32601,
  InvalidParams: -32602,
  InternalError: -32603,
} as const;

export function isRequest(msg: unknown): msg is Request {
  return (
    typeof msg === "object" &&
    msg !== null &&
    "id" in msg &&
    "method" in msg &&
    typeof (msg as { method: unknown }).method === "string"
  );
}

export function isNotification(msg: unknown): msg is Notification {
  return (
    typeof msg === "object" &&
    msg !== null &&
    !("id" in msg) &&
    "method" in msg &&
    typeof (msg as { method: unknown }).method === "string"
  );
}
```

```typescript
function streamFromString(input: string): Readable {
  return Readable.from([Buffer.from(input, "utf8")]);

class CapturingWritable extends Writable {
  chunks: string[] = [];
  _write(chunk: Buffer, _enc: BufferEncoding, cb: (err?: Error | null) => void) {
    this.chunks.push(chunk.toString("utf8"));
    cb();
  }

describe("readMessages", () => {
  it("yields each parsed JSON object on its own line", async () => {
    const stream = streamFromString(
      `{"id":1,"method":"a"}\n{"id":2,"method":"b"}\n`,
    );
    const collected: unknown[] = [];
    for await (const msg of readMessages(stream)) {
      collected.push(msg);
    }
    expect(collected).toEqual([
      { id: 1, method: "a" },
      { id: 2, method: "b" },
    ]);

    const stream = streamFromString(`not-json\n{"id":1,"method":"ok"}\n`);

      collected.push(msg);
    }

    expect(collected[0]).toEqual({ kind: "parse_error", line: "not-json" });
    expect(collected[1]).toEqual({ id: 1, method: "ok" });

  it("ignores blank lines", async () => {
    const stream = streamFromString(`\n{"id":1,"method":"a"}\n\n`);
    const collected: unknown[] = [];

      collected.push(msg);
    }
    expect(collected).toEqual([{ id: 1, method: "a" }]);

  it("handles input split across chunks (no trailing newline on last partial chunk)", async () => {
    const chunks = [`{"id":1,`, `"method":"a"}\n{"id":2,"method":"b"}\n`];
    const stream = Readable.from(chunks.map((c) => Buffer.from(c, "utf8")));
    const collected: unknown[] = [];
    for await (const msg of readMessages(stream)) {
      collected.push(msg);
    }
    expect(collected).toEqual([
      { id: 1, method: "a" },
      { id: 2, method: "b" },
    ]);
  });
});

describe("writeMessage", () => {
  it("writes JSON followed by a single LF", () => {
    const out = new CapturingWritable();
    writeMessage(out, { id: 1, result: {} });
    expect(out.chunks.join("")).toBe(`{"id":1,"result":{}}\n`);

  it("uses LF only (no CRLF)", () => {
```

```typescript
import type { Readable, Writable } from "node:stream";

export interface ParseError {
  kind: "parse_error";
  line: string;
}

function isParseError(v: unknown): v is ParseError {
  return (
    typeof v === "object" &&
    v !== null &&
    (v as { kind?: unknown }).kind === "parse_error"
  );
}

export async function* readMessages(
  stream: Readable,
): AsyncGenerator<unknown | ParseError> {
  let buffer = "";
  stream.setEncoding("utf8");
  for await (const chunk of stream) {
    buffer += chunk as string;
    let idx: number;
    while ((idx = buffer.indexOf("\n")) !== -1) {
      const line = buffer.slice(0, idx);
      buffer = buffer.slice(idx + 1);
      const trimmed = line.replace(/\r$/, "");
      if (trimmed.length === 0) continue;
      try {
        yield JSON.parse(trimmed);
      } catch {
        yield { kind: "parse_error", line: trimmed } satisfies ParseError;
      }
    }
  }
  // 終端: 残りバッファに有効な内容があれば最後に流す
  const tail = buffer.replace(/\r$/, "").trim();
  if (tail.length > 0) {
    try {
      yield JSON.parse(tail);
    } catch {
      yield { kind: "parse_error", line: tail } satisfies ParseError;
    }
  }
}

export function writeMessage(stream: Writable, msg: unknown): void {
  stream.write(JSON.stringify(msg) + "\n");
}

export { isParseError };
```

```typescript
import {
  ErrorCode,
  type IncomingMessage,
  type OutgoingMessage,
  isNotification,
  isRequest,
} from "./types.js";
import { isParseError } from "./transport.js";

export interface HandlerContext {
  sendNotification(method: string, params: unknown): void;
}

export type RequestHandler<P = unknown, R = unknown> = (
  params: P,
  ctx: HandlerContext,
) => Promise<R> | R;

export type NotificationHandler<P = unknown> = (
  params: P,
  ctx: HandlerContext,
) => Promise<void> | void;

export class Dispatcher {
  private requestHandlers = new Map<string, RequestHandler>();
  private notificationHandlers = new Map<string, NotificationHandler>();

  registerRequest<P = unknown, R = unknown>(
    method: string,
    handler: RequestHandler<P, R>,
  ): void {
    this.requestHandlers.set(method, handler as RequestHandler);
  }

  registerNotification<P = unknown>(
    method: string,
    handler: NotificationHandler<P>,
  ): void {
    this.notificationHandlers.set(method, handler as NotificationHandler);
  }

  async dispatch(
    message: unknown,
    ctx: HandlerContext,
  ): Promise<OutgoingMessage | null> {
    if (isParseError(message)) {
      return {
        id: null,
        error: { code: ErrorCode.ParseError, message: "Parse error" },
      };
    }

    if (isRequest(message)) {
      return await this.dispatchRequest(message, ctx);
    }

    if (isNotification(message)) {
      await this.dispatchNotification(message, ctx);
      return null;
    }

    return {
      id: null,
      error: { code: ErrorCode.InvalidRequest, message: "Invalid Request" },
    };
  }

  private async dispatchRequest(
    req: IncomingMessage & { id: unknown; method: string },
    ctx: HandlerContext,
  ): Promise<OutgoingMessage> {
    const handler = this.requestHandlers.get(req.method);
    const id = req.id as number | string;
    if (!handler) {
      return {
        id,
        error: {
          code: ErrorCode.MethodNotFound,
          message: `Method not found: ${req.method}`,
        },
      };
    }
    try {
      const result = await handler(req.params, ctx);
      return { id, result };
    } catch (err) {
      const message = err instanceof Error ? err.message : String(err);
      return {
        id,
        error: { code: ErrorCode.InternalError, message },
      };
    }
  }

  private async dispatchNotification(
    notif: IncomingMessage & { method: string },
    ctx: HandlerContext,
  ): Promise<void> {
    const handler = this.notificationHandlers.get(notif.method);
    if (!handler) return;
    try {
      await handler(notif.params, ctx);
    } catch (err) {
      const message = err instanceof Error ? err.message : String(err);
      process.stderr.write(
        `notification handler error (${notif.method}): ${message}\n`,
      );
    }
  }
}
```

- [ ] **Step 1: `apps/claude-app-server/src/server.ts` を作成**

```typescript
import { Dispatcher, type HandlerContext } from "./jsonrpc/dispatcher.js";
import { readMessages, writeMessage } from "./jsonrpc/transport.js";
import { registerHandlers } from "./handlers/index.js";

export interface ServerOptions {
  model?: string;
  permissionMode?: string;
  allowedTools?: string[];
}

export async function runServer(_opts: ServerOptions): Promise<void> {
  // CLAUDE.md: Pro/Max OAuth 認証を使うため API キー経路を遮断する
  delete process.env.ANTHROPIC_API_KEY;

  const dispatcher = new Dispatcher();
  registerHandlers(dispatcher);

  const ctx: HandlerContext = {
    sendNotification: (method, params) =>
      writeMessage(process.stdout, { method, params }),
  };

  for await (const message of readMessages(process.stdin)) {
    const response = await dispatcher.dispatch(message, ctx);
    if (response !== null) writeMessage(process.stdout, response);
  }
}
```

`_opts` 引数は Phase 1 では使わないが、Phase 2 で SDK 呼び出しに渡す予定なので受け取っておく。

```typescript
  .option(

    "default | acceptEdits | bypassPermissions",
  )

    "--allowed-tools <list>",
    "comma-separated tool allowlist",
    (v) => v.split(",").map((s) => s.trim()).filter(Boolean),

  )
  .parse();

const opts = program.opts() as {

  permissionMode?: string;
  allowedTools?: string[];
};

runServer(opts).catch((err: unknown) => {
```
