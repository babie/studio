# M3 Phase 1: `apps/conductor` Bootstrap + Memory Tracker Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Bootstrap the `apps/conductor` TypeScript package and prove an end-to-end happy path with a Memory tracker + Mock backend, where running `conductor examples/workflow.mock-memory.md` drives 2 fixture issues from `Todo → Done` in-process.

**Architecture:** Layered modules (`domain` / `config` / `tracker` / `backend` / `orchestrator` / `cli` / `util`) following kamae principles (discriminated unions, branded types, byethrow Result, valibot boundary validation, PII redact). The Orchestrator polls the Tracker for candidate issues, runs each through an AgentRunner that delegates a single turn to the Backend, and writes the resulting state back to the Tracker. Phase 1 only ships the in-process `Tracker.Memory` and `Backend.Mock`; real backends (subprocess + JSON-RPC) and Linear/GitHub trackers arrive in Phases 2-4.

**Tech Stack:** TypeScript 6 (NodeNext ESM, Node 20+), commander 14 (CLI), valibot 1 (schema), `@praha/byethrow` 0.11 (Result), `js-yaml` 4 + `@types/js-yaml` (YAML), vitest 4 (test runner, in-source tests via `import.meta.vitest`), oxlint / oxfmt (lint/format). Matches the `apps/claude-app-server` stack.

**Branch:** `feat/m3-phase1` (create at start of Pre-flight).

**Spec:** [`docs/superpowers/specs/2026-05-16-m3-phase1-design.md`](../specs/2026-05-16-m3-phase1-design.md)

    case "claude": {

        throw new Error("agent.type=claude requires a `claude:` block");

        approvalPolicy: yaml.codex.approval_policy,
        threadSandbox: yaml.codex.thread_sandbox,
        turnSandboxPolicy: yaml.codex.turn_sandbox_policy,

      return out;

    case "mock": {

        type: "mock",

        ...(block.force_fail !== undefined ? { forceFail: block.force_fail } : {}),

      return out;
    }

    activeStates: yaml.tracker.active_states,
    terminalStates: yaml.tracker.terminal_states,

        projectOwner: yaml.github.project_owner,
        projectNumber: yaml.github.project_number,
        ...(yaml.github.assignee !== undefined ? { assignee: yaml.github.assignee } : {}),
      };
      return out;
    }
  }
};

const buildWorkspace = (yaml: WorkflowYaml): WorkspaceConfig | undefined => {
  if (!yaml.workspace) return undefined;
  const hooks = yaml.workspace.hooks;
  return {
    root: yaml.workspace.root,
    ...(hooks
      ? {
          hooks: {
            ...(hooks.before_run !== undefined ? { beforeRun: hooks.before_run } : {}),
            ...(hooks.after_create !== undefined ? { afterCreate: hooks.after_create } : {}),
            ...(hooks.before_remove !== undefined ? { beforeRemove: hooks.before_remove } : {}),
          },
        }
      : {}),
  };
};

/**
 * Parses a raw YAML object (already loaded by js-yaml) and transforms it into
 * the strongly-typed `WorkflowConfig`. Does NOT attach the markdown prompt
 * body — caller (config/parser.ts) does that.
 *
 * Returns a byethrow Result; schema violations come back as a Failure with
 * StandardSchema issues. Cross-block invariants (e.g., agent.type=claude
 * requires a claude: block) throw inside the transform — callers should
 * wrap this in try/catch if they want to convert to Result.
 */
export const parseWorkflowYaml = (
  raw: unknown,
): Result.Result<Omit<WorkflowConfig, "prompt">, { issues: ReadonlyArray<v.BaseIssue<unknown>> }> => {
  const yamlResult = schemaResult(WorkflowYamlSchema)(raw);
  if (yamlResult.type === "Failure") {
    return yamlResult;
  }
  const yaml = yamlResult.value;
  const workspace = buildWorkspace(yaml);
  const out: Omit<WorkflowConfig, "prompt"> = {
    agent: buildAgent(yaml),
    tracker: buildTracker(yaml),
    ...(workspace !== undefined ? { workspace } : {}),
  };
  return { type: "Success", value: out };
};

import type { Result } from "@praha/byethrow";
import type { ConfigError } from "../domain/errors.js";

export type LoadedWorkflow = Readonly<{
  frontmatter: string;
  body: string;
}>;

const FRONT_DELIM = "---";

export const splitFrontmatter = (
  path: string,
  raw: string,
): Result.Result<LoadedWorkflow, ConfigError> => {
  const lines = raw.split("\n");
  if (lines[0] !== FRONT_DELIM) {
    return { type: "Failure", error: { kind: "frontmatter-missing", path } };
  }
  // Find the closing delimiter, starting from line 1.
  let closeIdx = -1;
  for (let i = 1; i < lines.length; i++) {
    if (lines[i] === FRONT_DELIM) {
      closeIdx = i;
      break;
    }
  }
  if (closeIdx === -1) {
    return { type: "Failure", error: { kind: "frontmatter-missing", path } };
  }
  const frontmatter = lines.slice(1, closeIdx).join("\n") + "\n";
  const body = lines.slice(closeIdx + 1).join("\n");
  return { type: "Success", value: { frontmatter, body } };
};

import type { MemoryTrackerConfig } from "../domain/tracker-config.js";
import type { Issue, IssueId, IssueStateName } from "../domain/issue.js";
import type { TrackerError } from "../domain/errors.js";
import type { Tracker } from "./types.js";

const normalize = (s: string): string => s.trim().toLowerCase();

export const createMemoryTracker = (config: MemoryTrackerConfig): Tracker => {
  const store: Issue[] = config.issues.map((i) => ({ ...i }));

  const findById = (id: IssueId): Issue | undefined => store.find((i) => i.id === id);

  const ok = <T>(value: T): Result.Result<T, TrackerError> => ({ type: "Success", value });

  return {
    fetchCandidateIssues: async () => ok(store.map((i) => ({ ...i }))),
    fetchIssuesByStates: async (states) => {
      const wanted = new Set(states.map(normalize));
      return ok(
        store
          .filter((i) => wanted.has(normalize(i.state)))
          .map((i) => ({ ...i })),
      );
    },
    fetchIssueStatesByIds: async (ids) => {
      const wanted = new Set(ids);
      return ok(store.filter((i) => wanted.has(i.id)).map((i) => ({ ...i })));
    },
    createComment: async () => ok(undefined),
    updateIssueState: async (issue, state) => {
      const target = findById(issue.id);
      if (!target) {
        return { type: "Failure", error: { kind: "issue-not-found", id: issue.id } };
      }
      target.state = state;
      return ok(undefined);
    },
  };
};

import * as v from "valibot";
import { IssueStateName } from "../domain/issue.js";
import type { ConfigError, OrchestratorError } from "../domain/errors.js";

export type RunCliOptions = Readonly<{ workflowPath: string }>;

const formatConfigError = (err: ConfigError): string => {
  switch (err.kind) {
    case "file-not-found":
      return `workflow file not found: ${err.path}`;
    case "frontmatter-missing":
      return `workflow frontmatter (--- ... ---) missing: ${err.path}`;
    case "yaml-parse-failed":
      return `YAML parse failed in ${err.path}: ${err.cause}`;
    case "schema-violation":
      return `schema violation in ${err.path}: ${err.issues.map((i) => i.message).join("; ")}`;
  }
};

const formatOrchestratorError = (err: OrchestratorError): string => {
  switch (err.kind) {
    case "config":
      return formatConfigError(err.error);
    case "tracker":
      return `tracker error (${err.error.kind})`;
    case "backend":
      return `backend error (${err.error.kind}): ${err.error.reason}`;
  }
};

export const runCli = async (opts: RunCliOptions): Promise<void> => {
  const logger = createStdLogger();
  const raw = await readFile(opts.workflowPath, "utf8").catch(() => null);
  if (raw === null) {
    logger.error(formatConfigError({ kind: "file-not-found", path: opts.workflowPath }));
    process.exit(1);
  }

  const parsed = parseWorkflow(opts.workflowPath, raw);
  if (parsed.type === "Failure") {
    logger.error(formatConfigError(parsed.error));
    process.exit(1);
  }
  const config = parsed.value;

  if (config.agent.backend.type !== "mock") {
    logger.error(
      `agent.type=${config.agent.backend.type} is not supported in Phase 1 (only "mock"). ` +
        `Redacted config: ${JSON.stringify(redactConfig(config))}`,
    );
    process.exit(2);
  }
  if (config.tracker.kind !== "memory") {
    logger.error(`tracker.kind=${config.tracker.kind} is not supported in Phase 1 (only "memory").`);
    process.exit(2);
  }

  const tracker = createMemoryTracker(config.tracker);
  const backend = createMockBackend(config.agent.backend);
  const result = await runOrchestrator({
    tracker,
    backend,
    promptTemplate: config.prompt,
    ...(config.tracker.doingState
      ? { doingState: v.parse(IssueStateName.schema, config.tracker.doingState) }
      : {}),
    ...(config.tracker.doneState
      ? { doneState: v.parse(IssueStateName.schema, config.tracker.doneState) }
      : {}),
    logger,
  });

  if (result.type === "Failure") {
    logger.error(formatOrchestratorError(result.error));
    process.exit(1);
  }

  logger.info(
    `[orchestrator] done: total=${result.value.total} completed=${result.value.completed} failed=${result.value.failed}`,

import { describe, it, expect } from "vitest";
import { parseWorkflow } from "../../src/config/parser.js";
import { createMemoryTracker } from "../../src/tracker/memory.js";
import { createMockBackend } from "../../src/backend/mock.js";
import { runOrchestrator } from "../../src/orchestrator/orchestrator.js";
import * as v from "valibot";
import { IssueStateName } from "../../src/domain/issue.js";
import { readFile } from "node:fs/promises";
import { fileURLToPath } from "node:url";
import { dirname, join } from "node:path";

const __dirname = dirname(fileURLToPath(import.meta.url));
const examplePath = join(__dirname, "../../examples/workflow.mock-memory.md");

describe("integration: mock-memory happy path", () => {
  it("drives 2 issues through Todo → Done", async () => {
    const raw = await readFile(examplePath, "utf8");
    const parsed = parseWorkflow(examplePath, raw);
    if (parsed.type !== "Success") throw new Error("expected schema success");

    const config = parsed.value;
    if (config.tracker.kind !== "memory") throw new Error("expected memory tracker");
    if (config.agent.backend.type !== "mock") throw new Error("expected mock backend");

    const tracker = createMemoryTracker(config.tracker);
    const backend = createMockBackend(config.agent.backend);
    const captured: string[] = [];

    const result = await runOrchestrator({
      tracker,
      backend,
      promptTemplate: config.prompt,
      ...(config.tracker.doingState
        ? { doingState: v.parse(IssueStateName.schema, config.tracker.doingState) }
        : {}),
      ...(config.tracker.doneState
        ? { doneState: v.parse(IssueStateName.schema, config.tracker.doneState) }
        : {}),
      logger: { info: (m) => captured.push(m), error: () => {} },
    });

    if (result.type !== "Success") throw new Error("expected success");
    expect(result.value.total).toBe(2);
    expect(result.value.completed).toBe(2);
    expect(result.value.failed).toBe(0);

    const final = await tracker.fetchIssuesByStates([v.parse(IssueStateName.schema, "Done")]);
    if (final.type !== "Success") throw new Error("unexpected tracker failure");
    expect(final.value.map((i) => i.identifier).sort()).toEqual(["MEMORY-1", "MEMORY-2"]);

    expect(captured.some((l) => l.includes("MEMORY-1 -> Done"))).toBe(true);
    expect(captured.some((l) => l.includes("MEMORY-2 -> Done"))).toBe(true);
  });
});
