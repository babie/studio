# M3 Phase 5: TUI dashboard + observability Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Port `apps/symphony/lib/symphony_elixir/status_dashboard.ex` (1,952 LoC) + `observability_pubsub.ex` to `apps/conductor` as a vanilla-ANSI TUI dashboard. byte-for-byte parity with symphony during the migration window (Phase 5 → 7). Phase 6 will populate parallel/retry state; Phase 7 renames "SYMPHONY STATUS" → "CONDUCTOR STATUS" and deletes the goldens.

**Architecture:** Three layers:
1. **State holder** (`ObservabilityState`): mutable per-process state. Implements `ObservabilityHooks`. Orchestrator / AgentRunner / BackendSession push events through hooks; state mutates synchronously.
2. **Event bus** (`EventBus`): typed `EventEmitter` wrapper. `state.attachBus(bus)` makes mutators `emit("stateChanged")` with a 16ms debounce so high-frequency `item/agentMessage/delta` doesn't peg the loop.
3. **Dashboard driver** (`Dashboard`): tick timer (`refresh_ms`=1000) + bus subscribe; pulls snapshot, computes `tps` from token samples, renders via pure `formatSnapshot()`, throttles writes via `render_interval_ms`=16 + `MINIMUM_IDLE_RERENDER_MS`=1000 + `lastSnapshotFingerprint`/`lastRenderedContent` skip.

Renderer is split into pure sub-modules (`render/{format,sparkline,header,project-link,running-table,backoff-queue,offline,snapshot}.ts`). ANSI escape sequences are direct strings (no `chalk`/`picocolors`) so output matches symphony's `IO.ANSI.*` byte-for-byte. The CLI starts/stops the dashboard with `--no-dashboard` (forces OFF) and `CONDUCTOR_DEBUG=1` (forces OFF + restores stderr logger).

Symphony fixtures are captured **once** by an Elixir script (`apps/symphony/scripts/capture_dashboard_fixtures.exs`) at Task 1; output is committed as `.txt` (golden) + `.json` (snapshot spec) pairs. Conductor's `dashboard-parity.test.ts` reads the same `.json`, renders via `formatSnapshot()`, asserts byte equality vs the `.txt`. Phase 7 deletes both the goldens and the capture script.

**Tech Stack:** Phase 4 stack. New: Node std `events.EventEmitter`. No new npm deps.

**Branch:** `feat/m3-phase5` (create at start of Pre-flight; user requested branch, not worktree).

**Spec:** [`docs/superpowers/specs/2026-05-17-m3-phase5-design.md`](../specs/2026-05-17-m3-phase5-design.md)

```ts
  cursorShow: "\x1b[?25h",
} as const;

export const colorize = (text: string, color: string): string =>
  `${color}${text}${ANSI.reset}`;

/** Format a count as "1.2k", "1.2M", or the raw integer (matches symphony format_count/1). */
export const formatCount = (n: number): string => {
  if (n < 1_000) return `${n}`;
  if (n < 1_000_000) {
    const v = n / 1_000;
    return `${v.toFixed(v >= 10 ? 0 : 1)}k`;
  }
  const v = n / 1_000_000;
  return `${v.toFixed(v >= 10 ? 0 : 1)}M`;
};

/** Format runtime seconds (>= 0) as "Xs", "Xm Ys", or "Xh Ym" (matches symphony format_runtime_seconds/1). */
export const formatRuntimeSeconds = (s: number): string => {
  const sec = Math.max(0, Math.floor(s));
  if (sec < 60) return `${sec}s`;
  if (sec < 3600) {
    const m = Math.floor(sec / 60);
    const r = sec % 60;
    return `${m}m ${r}s`;
  }
  const h = Math.floor(sec / 3600);
  const m = Math.floor((sec % 3600) / 60);
  return `${h}h ${m}m`;
};

/** Format tps as "12.3" (one decimal), or "0" when zero. */
export const formatTps = (tps: number): string =>
  tps === 0 ? "0" : tps.toFixed(1);

/** Truncate a string to max display width, appending "…" if shortened.
 *  Display width is measured in code points (close enough for ASCII + CJK). */
export const truncate = (s: string, max: number): string => {
  if (max <= 0) return "";
  const arr = Array.from(s);
  if (arr.length <= max) return s;
  if (max === 1) return "…";
  return arr.slice(0, max - 1).join("") + "…";
};
```

```ts
// apps/conductor/src/observability/render/sparkline.ts
const BLOCKS = ["▁", "▂", "▃", "▄", "▅", "▆", "▇", "█"] as const;
const COLUMNS = 24;

/** Sample of [timestampMs, cumulativeTotalTokens]. */
export type TokenSample = readonly [number, number];

/** Build a sparkline for the throughput window.
 *  @param samples newest-first or oldest-first; we sort internally.
 *  @param now wall clock ms (used as the right edge of the graph)
 *  @param windowMs total graph window (e.g. 600_000)
 *  Returns exactly COLUMNS characters. Empty / monotonic-flat windows return all ▁. */
export const sparkline = (
  samples: ReadonlyArray<TokenSample>,
  now: number,
  windowMs: number,
): string => {
  if (samples.length < 2) return BLOCKS[0].repeat(COLUMNS);
  const sorted = [...samples].sort((a, b) => a[0] - b[0]);
  const start = now - windowMs;
  const bucketMs = windowMs / COLUMNS;
  const deltas: number[] = new Array(COLUMNS).fill(0);
  for (let i = 1; i < sorted.length; i++) {
    const [t, tokens] = sorted[i]!;
    if (t < start || t > now) continue;
    const prev = sorted[i - 1]!;
    const diff = Math.max(0, tokens - prev[1]);
    const bucket = Math.min(COLUMNS - 1, Math.floor((t - start) / bucketMs));
    deltas[bucket]! += diff;
  }
  const max = Math.max(...deltas);
  if (max === 0) return BLOCKS[0].repeat(COLUMNS);
  return deltas.map((d) => BLOCKS[Math.min(BLOCKS.length - 1, Math.round((d / max) * (BLOCKS.length - 1)))]!).join("");
};

/** Prune samples older than `now - windowMs` and append the new (now, tokens) sample. */
export const updateTokenSamples = (
  samples: ReadonlyArray<TokenSample>,
  now: number,
  totalTokens: number,
  windowMs: number,
): ReadonlyArray<TokenSample> => {
  const cutoff = now - windowMs;
  const kept = samples.filter(([t]) => t >= cutoff);
  return [...kept, [now, totalTokens]];
};

/** Throughput (tokens/sec) averaged over the window's last `windowMs` slice.
 *  Caches by `now` second to mirror symphony's throttled_tps. */
export const throttledTps = (
  lastSecond: number | null,
  lastValue: number | null,
  now: number,
  samples: ReadonlyArray<TokenSample>,
  windowMs: number,
): { second: number; value: number } => {
  const second = Math.floor(now / 1000);
  if (lastSecond === second && lastValue !== null) return { second, value: lastValue };
  if (samples.length < 2) return { second, value: 0 };
  const sorted = [...samples].sort((a, b) => a[0] - b[0]);
  const cutoff = now - windowMs;
  const recent = sorted.filter(([t]) => t >= cutoff);
  if (recent.length < 2) return { second, value: 0 };
  const first = recent[0]!;
  const last = recent[recent.length - 1]!;
  const dt = Math.max(1, last[0] - first[0]);
  const dv = Math.max(0, last[1] - first[1]);
  return { second, value: (dv / dt) * 1000 };
};

if (import.meta.vitest) {
  const { describe, it, expect } = import.meta.vitest;
  describe("observability/render/sparkline", () => {
    it("empty samples → all ▁", () => {
      expect(sparkline([], 1000, 600_000)).toBe("▁".repeat(24));
    });
    it("single sample → all ▁", () => {
      expect(sparkline([[1000, 100]], 1000, 600_000)).toBe("▁".repeat(24));
    });
    it("monotonic constant → all ▁", () => {
      const s: TokenSample[] = [[1000, 5], [2000, 5], [3000, 5]];
      expect(sparkline(s, 3000, 600_000)).toBe("▁".repeat(24));
    });
    it("rising series ends with higher block", () => {
      const s: TokenSample[] = Array.from({ length: 12 }, (_, i) => [i * 50_000, i * 100]);
      const out = sparkline(s, 600_000, 600_000);
      expect(out.length).toBe(24);
      expect(out[out.length - 1]).toBe("█");
    });
    it("updateTokenSamples prunes old and appends", () => {
      const s: TokenSample[] = [[100, 5], [200, 6]];
      const next = updateTokenSamples(s, 700_000, 10, 600_000);
      expect(next).toEqual([[700_000, 10]]);
    });
    it("throttledTps caches per second", () => {
      const s: TokenSample[] = [[1000, 0], [6000, 50]];
      const r1 = throttledTps(null, null, 6000, s, 5_000);
      const r2 = throttledTps(r1.second, r1.value, 6500, s, 5_000);
      expect(r2.value).toBe(r1.value);
    });
  });
}
```

```ts
    issueId: string, identifier: string, state: string,
    pid: number | null, workspacePath: string | null,
  ): void {
    const now = new Date();
    if (this.startTimeMs == null) this.startTimeMs = now.getTime();
    const existing = this.running.get(issueId);
    const entry: RunningEntry = {
      issueId, identifier, state,
      workerHost: null, workspacePath,

      runtimeSeconds: existing?.runtimeSeconds ?? 0,

    this.bus?.emit("stateChanged");
  }

    this.running.delete(issueId);
    this.retrying.delete(issueId);
    this.bus?.emit("stateChanged");
  }

    const entry = this.running.get(issueId);
    if (!entry) return;
    const now = new Date();

    switch (event.kind) {
      case "thread-started":
        next = { ...next, sessionId: event.sessionId, lastCodexEvent: "thread started" };
        break;
      case "turn-started":
        next = { ...next, turnCount: next.turnCount + 1, lastCodexEvent: "turn started" };
        break;
      case "item-started":
        next = { ...next, lastCodexEvent: event.itemType };
        break;
      case "agent-message":
        next = { ...next, lastCodexMessage: previewMessage(event.text), lastCodexEvent: "agent message" };
        break;
      case "command-execution":
        next = { ...next, lastCodexEvent: `cmd: ${event.command}` };
        break;
      case "file-change":
        next = { ...next, lastCodexEvent: `edit: ${event.path}` };
        break;
      case "item-completed":
        next = { ...next, lastCodexEvent: "item done" };
        break;
      case "turn-completed":
        if (event.usage) {
          next = {
            ...next,
            codexInputTokens: next.codexInputTokens + event.usage.inputTokens,
            codexOutputTokens: next.codexOutputTokens + event.usage.outputTokens,
            codexTotalTokens: next.codexTotalTokens + event.usage.totalTokens,
          };
          this.codexTotals = {
            inputTokens: this.codexTotals.inputTokens + event.usage.inputTokens,
            outputTokens: this.codexTotals.outputTokens + event.usage.outputTokens,
            totalTokens: this.codexTotals.totalTokens + event.usage.totalTokens,
            secondsRunning: this.codexTotals.secondsRunning,
          };
        }
        next = { ...next, lastCodexEvent: "turn done" };
        break;
    }
    this.running.set(issueId, next);
    this.bus?.emit("stateChanged");
  }

    const existing = this.retrying.get(issueId);

    this.retrying.set(issueId, {

      workerHost: null,
      workspacePath: this.running.get(issueId)?.workspacePath ?? null,
    });
    this.bus?.emit("stateChanged");
  }

    this.retrying.delete(issueId);
    this.bus?.emit("stateChanged");

    this.polling = { checking, nextPollInMs, pollIntervalMs: intervalMs };
    this.bus?.emit("stateChanged");
  }
}
```

```ts
// apps/conductor/src/observability/event-bus.ts
import { EventEmitter } from "node:events";

export type EventName = "stateChanged";

const DEFAULT_DEBOUNCE_MS = 16;

/** Typed wrapper around Node EventEmitter with a debounce so high-frequency
 *  emit() calls coalesce into one delivery per debounce window. */
export class EventBus {
  private readonly emitter = new EventEmitter();
  private debounceTimer: NodeJS.Timeout | null = null;
  private readonly debounceMs: number;
  private readonly pending = new Set<EventName>();

  constructor(debounceMs: number = DEFAULT_DEBOUNCE_MS) {
    this.debounceMs = debounceMs;
  }

  emit(event: EventName): void {
    this.pending.add(event);
    if (this.debounceTimer != null) return;
    this.debounceTimer = setTimeout(() => {
      const drained = Array.from(this.pending);
      this.pending.clear();
      this.debounceTimer = null;
      for (const e of drained) this.emitter.emit(e);
    }, this.debounceMs);
  }

  on(event: EventName, handler: () => void): () => void {
    this.emitter.on(event, handler);
    return () => this.emitter.off(event, handler);
  }

  /** Cancel any pending debounced emits. Used by Dashboard.stop(). */
  flush(): void {
    if (this.debounceTimer != null) {
      clearTimeout(this.debounceTimer);
      this.debounceTimer = null;
      this.pending.clear();
    }
  }
}

if (import.meta.vitest) {
  const { describe, it, expect, vi } = import.meta.vitest;
  describe("observability/event-bus", () => {
    it("coalesces multiple emits within the debounce window", async () => {
      const bus = new EventBus(10);
      let count = 0;
      bus.on("stateChanged", () => { count++; });
      bus.emit("stateChanged");
      bus.emit("stateChanged");
      bus.emit("stateChanged");
      expect(count).toBe(0);
      await new Promise((r) => setTimeout(r, 25));
      expect(count).toBe(1);
    });
    it("off() unsubscribes", async () => {
      const bus = new EventBus(5);
      let count = 0;
      const off = bus.on("stateChanged", () => { count++; });
      off();
      bus.emit("stateChanged");
      await new Promise((r) => setTimeout(r, 15));
      expect(count).toBe(0);
    });
    it("flush cancels pending without firing handler", async () => {
      const bus = new EventBus(50);
      const handler = vi.fn();
      bus.on("stateChanged", handler);
      bus.emit("stateChanged");
      bus.flush();
      await new Promise((r) => setTimeout(r, 70));
      expect(handler).not.toHaveBeenCalled();
    });
  });
}
```

```ts
  state: ObservabilityState;
  bus: EventBus;
  refreshMs: number;
  renderIntervalMs: number;
  runtimeContext: FormatRuntimeContext;

  getTerminalColumns?: () => number;
  /** Override for testing — defaults to Date.now. */
  now?: () => number;

const totalTokens = (snap: ReturnType<ObservabilityState["getSnapshot"]>): number =>
  snap.type === "ok" ? snap.data.codexTotals.totalTokens : 0;

export class Dashboard {
  private readonly opts: DashboardOptions;
  private readonly renderFn: (s: string) => void;
  private readonly getTerminalColumns: () => number;
  private readonly now: () => number;
  private tickTimer: NodeJS.Timeout | null = null;
  private flushTimer: NodeJS.Timeout | null = null;
  private tokenSamples: ReadonlyArray<TokenSample> = [];
  private lastTpsSecond: number | null = null;
  private lastTpsValue: number | null = null;
  private lastRenderedContent: string | null = null;
  private lastRenderedAtMs: number | null = null;
  private pendingContent: string | null = null;
  private lastSnapshotFingerprint: string | null = null;
  private busOff: (() => void) | null = null;
  private running = false;

  constructor(opts: DashboardOptions) {
    this.opts = opts;
    this.renderFn = opts.renderFn ?? ((s) => process.stdout.write(s));
    this.getTerminalColumns = opts.getTerminalColumns ?? (() => process.stdout.columns ?? DEFAULT_TERMINAL_COLUMNS);
    this.now = opts.now ?? Date.now;
  }

  start(): void {
    if (this.running) return;
    this.running = true;
    this.renderFn(ANSI.cursorHide);
    this.renderNow();
    this.tickTimer = setInterval(() => this.renderNow(), this.opts.refreshMs);
    this.busOff = this.opts.bus.on("stateChanged", () => this.renderNow());
  }

  stop(): void {
    if (!this.running) return;
    this.running = false;
    if (this.tickTimer) { clearInterval(this.tickTimer); this.tickTimer = null; }
    if (this.flushTimer) { clearTimeout(this.flushTimer); this.flushTimer = null; }
    if (this.busOff) { this.busOff(); this.busOff = null; }
    this.opts.bus.flush();
    if (this.pendingContent != null) this.writeRender(this.pendingContent, this.now());
    this.renderFn(ANSI.cursorShow + "\n");
  }

  private renderNow(): void {
    const now = this.now();
    const snapshot = this.opts.state.getSnapshot(now);
    this.tokenSamples = updateTokenSamples(this.tokenSamples, now, totalTokens(snapshot), THROUGHPUT_GRAPH_WINDOW_MS);
    const { second, value: tps } = throttledTps(
      this.lastTpsSecond, this.lastTpsValue, now, this.tokenSamples, THROUGHPUT_WINDOW_MS,
    );
    this.lastTpsSecond = second;
    this.lastTpsValue = tps;
    const fingerprint = JSON.stringify(snapshot);
    const idleRerender =
      this.lastRenderedAtMs != null && now - this.lastRenderedAtMs >= MINIMUM_IDLE_RERENDER_MS;
    if (fingerprint === this.lastSnapshotFingerprint && !idleRerender) return;
    this.lastSnapshotFingerprint = fingerprint;
    const content = formatSnapshot(snapshot, tps, this.getTerminalColumns(), this.opts.runtimeContext);
    this.enqueueRender(content, now);
  }

  private enqueueRender(content: string, now: number): void {
    if (content === this.lastRenderedContent) return;
    if (this.canRenderNow(now)) {
      this.writeRender(content, now);
    } else {
      this.pendingContent = content;
      this.scheduleFlush(now);
    }
  }

  private canRenderNow(now: number): boolean {
    if (this.lastRenderedAtMs == null) return true;
    return now - this.lastRenderedAtMs >= this.opts.renderIntervalMs;
  }

  private scheduleFlush(now: number): void {
    if (this.flushTimer != null) return;
    const since = this.lastRenderedAtMs == null ? this.opts.renderIntervalMs : now - this.lastRenderedAtMs;
    const delay = Math.max(1, this.opts.renderIntervalMs - since);
    this.flushTimer = setTimeout(() => {
      this.flushTimer = null;
      if (this.pendingContent != null) {
        this.writeRender(this.pendingContent, this.now());
        this.pendingContent = null;
      }
    }, delay);
  }

  private writeRender(content: string, now: number): void {
    this.renderFn(ANSI.clearAndHome + content + "\n");
    this.lastRenderedContent = content;
    this.lastRenderedAtMs = now;
  }
}
```

```ts
// apps/conductor/scripts/diff-symphony-dashboard.ts
// Dev-only: capture symphony output for each fixture spec, compare to conductor
// formatSnapshot output, print colored diff. Removed in Phase 7.

import { readdirSync, readFileSync } from "node:fs";
import { spawnSync } from "node:child_process";
import { join } from "node:path";
import { formatSnapshot } from "../src/observability/render/snapshot.js";
import type { ObservabilitySnapshot } from "../src/domain/observability-snapshot.js";

const FIXTURES_DIR = join(__dirname, "..", "test", "fixtures", "dashboard");
const SYMPHONY_DIR = join(__dirname, "..", "..", "symphony");

const main = () => {
  const json = readdirSync(FIXTURES_DIR).filter((f) => f.endsWith(".json"));
  let failures = 0;
  for (const f of json) {
    const name = f.replace(/\.json$/, "");
    const spec = JSON.parse(readFileSync(join(FIXTURES_DIR, f), "utf8"));
    const symResult = spawnSync(
      "mix",
      ["run", "-e", `IO.write(SymphonyElixir.StatusDashboard.__capture__(${JSON.stringify(spec.snapshot)}, ${spec.tps}, ${spec.terminalColumns}))`],
      { cwd: SYMPHONY_DIR, encoding: "utf8" },
    );
    const symphonyOut = symResult.stdout ?? "";
    // hydrate snapshot dates (same logic as parity test)
    const snap: ObservabilitySnapshot = spec.snapshot.type === "error"
      ? { type: "error" }
      : { type: "ok", data: {
          ...spec.snapshot.data,
          running: spec.snapshot.data.running.map((r: Record<string, unknown>) => ({
            ...r,
            startedAt: new Date(r.startedAt as string),
            lastCodexTimestamp: r.lastCodexTimestamp ? new Date(r.lastCodexTimestamp as string) : null,
          })),
        }};
    const conductorOut = formatSnapshot(snap, spec.tps, spec.terminalColumns, {
      projectLink: { kind: "memory" }, maxAgents: spec.runtimeContext?.maxAgents ?? 2,
    });
    if (symphonyOut === conductorOut + "\n" || symphonyOut === conductorOut) {
      console.log(`PASS ${name}`);
    } else {
      failures++;
      console.log(`FAIL ${name}`);
      console.log("--- symphony");
      console.log(JSON.stringify(symphonyOut));
      console.log("--- conductor");
      console.log(JSON.stringify(conductorOut));
    }
  }
  if (failures > 0) process.exit(1);
};

main();
```
